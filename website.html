const express = require("express"); const path = require("path"); const app = express(); const PORT = 3000; // Only allow requests to this API host. const ALLOWED_HOST = "jsonplaceholder.typicode.com"; app.use(express.static(path.join(__dirname, "public"))); app.get("/proxy/*", async (req, res) => { try { const targetPath = req.params[0]; const targetUrl = `https://${ALLOWED_HOST}/${targetPath}`; const response = await fetch(targetUrl); if (!response.ok) { return res.status(response.status).json({ error: `Upstream returned ${response.status}` }); } const contentType = response.headers.get("content-type") || ""; if (contentType.includes("application/json")) { const data = await response.json(); return res.json(data); } const text = await response.text(); res.type("text/plain").send(text); } catch (err) { res.status(500).json({ error: "Proxy request failed" }); } }); app.listen(PORT, () => { console.log(`Running at http://localhost:${PORT}`); });
